from pptx import Presentation
from pptx.util import Pt

# 1. יצירת אובייקט המצגת
prs = Presentation()

# 2. בחירת סוג השקף (0 = שקף כותרת ראשי)
title_slide_layout = prs.slide_layouts[0]
slide = prs.slides.add_slide(title_slide_layout)

# 3. גישה לתיבות הטקסט בשקף
title = slide.shapes.title
subtitle = slide.placeholders[1]

# 4. הזרקת התוכן שלנו
title.text = "פיתוח ומימוש בקר VGA על FPGA"
subtitle.text = "הגנת פרויקט גמר\nמגישים: יוסי ברים ואלעד אסבג"

# 5. שמירת הקובץ בתיקייה
output_path = "VGA_Presentation_Approved.pptx"
prs.save(output_path)

print(f"המצגת נשמרה בהצלחה בשם: {output_path}")
